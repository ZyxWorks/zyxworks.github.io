# freeze any already-indexed URL in place forever.

# Everyone else, AI crawlers included (GPTBot, ClaudeBot, Google-Extended,
# PerplexityBot, CCBot, Bytespider, ...): stay out, all paths.
User-agent: *
Disallow: /
